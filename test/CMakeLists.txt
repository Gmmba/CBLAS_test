cmake_minimum_required(VERSION 3.15)
project(SYMM_Benchmark C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fopenmp -O3")
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENBLAS REQUIRED openblas)
add_executable(benchmark_symm benchmark_symm.c symm.c)
target_include_directories(benchmark_symm PRIVATE ${OPENBLAS_INCLUDE_DIRS})
target_link_libraries(benchmark_symm ${OPENBLAS_LIBRARIES} m)
