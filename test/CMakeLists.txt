cmake_minimum_required(VERSION 3.15)
project(CBLAS_Tests C)
set(CMAKE_C_STANDARD 11)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENBLAS REQUIRED openblas)
add_executable(tests_cblas tests.c)
target_include_directories(tests_cblas PRIVATE ${OPENBLAS_INCLUDE_DIRS})
target_link_libraries(tests_cblas ${OPENBLAS_LIBRARIES})
target_compile_options(tests_cblas PRIVATE ${OPENBLAS_CFLAGS_OTHER})
